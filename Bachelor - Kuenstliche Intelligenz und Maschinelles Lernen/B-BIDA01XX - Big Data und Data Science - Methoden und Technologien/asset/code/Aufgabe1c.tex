def convert_gross(value):
    """
    Konvertiert Strings wie '$857,190,335' in Zahlen (Float).
    """
    if isinstance(value, str):
        # $, M und Kommas entfernen
        value = value.replace('$', '').replace('M', '').replace(',', '')
        # Zu float konvertieren
        return float(value)
    return value
