% Configuration for code listings and syntax highlighting

% Color definitions for code syntax highlighting
\definecolor{codegray}{gray}{0.9}           % Light gray for frame background
\definecolor{codeblue}{RGB}{0, 84, 150}     % Corporate blue for keywords
\definecolor{codegreen}{rgb}{0,0.6,0}       % Green for comments
\definecolor{codered}{rgb}{0.6,0,0}         % Red for strings
\definecolor{primary}{RGB}{0, 84, 150}      % Primary corporate color
\definecolor{secondary}{RGB}{0, 84, 150}    % Secondary corporate color

% Global listing settings (apply to all code listings)
\lstset{
    basicstyle=\ttfamily\footnotesize,      % Monospace font, small size
    frame=single,                           % Frame around code
    framesep=5pt,                           % Padding inside frame
    framerule=1pt,                          % Frame border thickness
    rulecolor=\color{codegray},             % Frame color
    backgroundcolor=\color{white!95!black}, % 95% white, 5% black = very light gray
    breaklines=true,                        % Automatic line breaking
    showstringspaces=false,                 % Hide spaces in strings
    tabsize=4,                              % Tab width = 4 spaces
    numbers=left,                           % Line numbers on left
    numberstyle=\tiny\color{gray},          % Line number style
    numbersep=11pt,                         % Space between numbers and code
    captionpos=b,                           % Caption position (bottom)
    abovecaptionskip=11pt,                  % Space above caption
    belowcaptionskip=11pt,                  % Space below caption
    inputencoding=utf8,                     % UTF-8 character encoding
    literate=                               % German character support (for code comments)
    {ä}{{\"a}}1 {ö}{{\"o}}1 {ü}{{\"u}}1     % Lowercase umlauts
    {Ä}{{\"A}}1 {Ö}{{\"O}}1 {Ü}{{\"U}}1     % Uppercase umlauts
    {ß}{{\ss}}1                             % Sharp S
}

% Python-specific syntax highlighting
